用 fetch 获取接口数据

页面加载后向猎犬随机图片接口发起 GET 请求，拿到 JSON 响应后解析出图片地址，再用原生 JavaScript 渲染成图片卡片。点击「重新请求」可再次调用接口，观察加载、成功与失败三种状态。

本页使用的请求代码原生 fetch + async / await 
// 1. 接口地址（该接口允许跨域访问，可直接在页面中请求）
var ghz_apiUrl = 'https://dog.ceo/api/breed/hound/images/random/3';

// 2. 发起请求并解析 JSON
async function ghz_loadData() {
    var ghz_response = await fetch(ghz_apiUrl);

    // 3. HTTP 状态码不是 2xx 时抛出错误
    if (!ghz_response.ok) {
        throw new Error('HTTP ' + ghz_response.status);
    }

    // 4. 把响应体转成 JS 对象
    var ghz_data = await ghz_response.json();

    // 5. ghz_data.message 是图片地址数组，ghz_data.status 为 'success'
    ghz_render(ghz_data.message);
}
