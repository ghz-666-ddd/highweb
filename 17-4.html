// 1. 组装请求配置：方法、请求头与请求体
var ghz_options = {
    method: ghz_method,
    headers: { 'Content-Type': 'application/json' },
    body: ghz_method === 'GET' ? null : ghz_body
};

// 2. 发起请求
var ghz_response = await fetch(ghz_url, ghz_options);

// 3. 读取响应正文并尝试解析为 JSON
var ghz_text = await ghz_response.text();
try {
    ghz_data = JSON.parse(ghz_text);
} catch (ghz_error) {
    // 不是 JSON 时按纯文本展示
}
